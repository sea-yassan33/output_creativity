使用方法について

例はすべて下のサンプル df を前提にしています。コードをクリックするとコピーできます。

import pandas as pd
import numpy as np

df = pd.DataFrame({
    "name":  ["佐藤", "鈴木", "高橋", "田中"],
    "dept":  ["営業", "開発", "営業", "開発"],
    "sales": [120, 80, np.nan, 150],
    "date":  ["2026-04-01", "2026-04-15", "2026-05-01", "2026-05-20"],
})
