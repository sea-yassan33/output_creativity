# 仮想環境の使用方法

https://qiita.com/fiftystorm36/items/b2fd47cf32c7694adc2e

```sh
## プロジェクトディレクトリへ移動
cd my-project

## 仮想環境を作成
python -m venv <仮想環境名>
python -m venv .venv

## 仮想環境を有効化（activate）
source .venv/Scripts/activate

## 仮想環境の無効化
deactivate

## requirements.txt を使った管理
pip freeze > requirements.txt

## 別環境で再現
python -m venv venv
source venv/bin/activate
pip install -r requirements.txt

## 仮想環境を作り直す
deactivate
rm -rf venv
python -m venv venv
```

## ライブラリー確認スクリプト

```py
import importlib.util
import sys
# ここに「必要な外部ライブラリ」を書く
# キー: importするときの名前 / 値: pip install するときの名前
# （多くは同じだが、例: import cv2 → pip install opencv-python のように違うこともある）
REQUIRED = {
    "requests": "requests",
    "pandas": "pandas",
    "cv2": "opencv-python",
}
def check_dependencies():
    missing = []
    for import_name, pip_name in REQUIRED.items():
        if importlib.util.find_spec(import_name) is None:
            missing.append(pip_name)
    if missing:
        print("─" * 20)
        print("必要なライブラリがインストールされていません。")
        print("以下のライブラリが不足しています:")
        for name in missing:
            print(f"  - {name}")
        print()
        print("次のコマンドを実行してインストールしてください:")
        print()
        print(f"  {sys.executable} -m pip install {' '.join(missing)}")
        print()
        print("インストール後、もう一度このプログラムを実行してください。")
        print("─" * 20)
        sys.exit(1)  # 本体を実行せずにここで終了
# ライブラリのチェックを最初に行う
check_dependencies()

# --- ここから下が本体 ---
# チェックを通過した後で初めて外部ライブラリを import する
import requests
import pandas as pd

def main():
    print("本体プログラムを実行します")
    # ... 実際の処理 ...
if __name__ == "__main__":
    main()
```
